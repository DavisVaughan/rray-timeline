  r_obj* x,
  r_obj* y,

  struct rray_arg* x_arg,
  struct rray_arg* y_arg,
  struct r_lazy error_call
) {

  check_unclassed(y, y_arg, error_call);

  x = KEEP(arg_as_array(x, x_arg, error_call));
  y = KEEP(arg_as_array(y, y_arg, error_call));

  const rray_equality_fn fn =
    rray_equality_switch(x, y, op, x_arg, y_arg, error_call);

  r_obj* x_dimensions = r_dim(x);
  r_obj* y_dimensions = r_dim(y);

  const int* v_x_dimensions = r_int_cbegin(x_dimensions);
  const int* v_y_dimensions = r_int_cbegin(y_dimensions);

  const int x_dimensionality =
    rray_dimensionality_from_dimensions(x_dimensions);
  const int y_dimensionality =
    rray_dimensionality_from_dimensions(y_dimensions);

  r_obj* dimensions = KEEP(rray_dimensions2(
    v_x_dimensions,
    x_dimensionality,
    v_y_dimensions,
    y_dimensionality,
    x_arg,
    y_arg,
    error_call
  ));
  const int* v_dimensions = r_int_cbegin(dimensions);
  const int dimensionality = rray_dimensionality_from_dimensions(dimensions);

  struct rray_strided_iterator2_plan plan = rray_broadcast_iterator2_plan(
    v_x_dimensions,
    x_dimensionality,
    v_y_dimensions,
    y_dimensionality,
    v_dimensions,
    dimensionality
  );

  r_obj* out = KEEP(fn(x, y, &plan, op));
  r_attrib_poke_dim(out, dimensions);

  r_obj* out_names = KEEP(rray_broadcast_names2(x, y, dimensions));

  if (out_names != r_null) {
    r_attrib_poke_dim_names(out, out_names);
  }

  FREE(5);
  return out;
}

Type dispatch in rray4 is written as an exhaustive `switch`, following vctrs'
`vec_typeof2()` and `vec_ptype2_switch_native()`. Build a symmetric combined
enum with one entry per unordered pair of native types, a `rray_typeof2()` that
maps a pair of `enum rray_type` onto it, then switch over every entry.

Do not collapse this into a rank function, a max, a table lookup, or any other
arithmetic shortcut, even though it is shorter.

`enum rray_type` is restricted to the native types on purpose, so these switches
need no `default` and the compiler reports a missing case.

**Why:** The switch names every combination, so an unhandled or wrong pair is
visible in the source rather than implied by a formula. It is also where new
type rules get added.

**Value:** This applies again to `rray_binary_ptype()` and
`rray_reduction_ptype()` and their promotion tables.

**How to apply:** Pair the switch with an exhaustive test. `helper-ptype.R` has
`native_ptype_matrix()`, which walks all 7x7 pairs and snapshots the result type
(`NA` where the pair errors), so a typo in any arm shows up in the snapshot.
