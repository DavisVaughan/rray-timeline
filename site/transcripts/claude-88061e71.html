static void check_split_dimensions(
  r_obj* dimensions,
  int axis_dimension,
  struct rray_arg* arg,
  struct r_lazy error_call
) {
  const r_ssize size = r_length(dimensions);
  const int* v_dimensions = r_int_cbegin(dimensions);

  if (size == 1) {
    const int dimension = v_dimensions[0];

    if (dimension == 0) {
      r_abort_lazy_call(
        error_call,
        "A single %s value must be positive, not 0.",
        rray_arg_format(arg)
      );
    }

    if (axis_dimension % dimension != 0) {
      r_abort_lazy_call(
        error_call,
        "A single %s value of %d must evenly divide the `axis` dimension "
        "of %d.",
        rray_arg_format(arg),
        dimension,
        axis_dimension
      );
    }

    return;
  }

  r_ssize total = 0;

  for (r_ssize i = 0; i < size; ++i) {
    total += v_dimensions[i];
  }

  if (total != axis_dimension) {
    r_abort_lazy_call(
      error_call,
      "%s must sum to the `axis` dimension of %d, not %" R_PRI_SSIZE ".",
      rray_arg_format(arg),
      axis_dimension,
      total
    );
  }
}
