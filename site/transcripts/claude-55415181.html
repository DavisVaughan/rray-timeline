test_that("double NA / NaN propagates", {
  x <- c(1, NA_real_)
  expect_identical(as.vector(rray_sum(x, 1L)), NA_real_)

  x <- c(NA_real_, 1)
  expect_identical(as.vector(rray_sum(x, 1L)), NA_real_)

  x <- c(1, NaN)
  expect_identical(as.vector(rray_sum(x, 1L)), NaN)

  x <- c(NaN, 1)
  expect_identical(as.vector(rray_sum(x, 1L)), NaN)

  x <- c(NA, NaN)
  expect_identical(is.na(rray_sum(x, 1L)), array(TRUE, 1L))

  x <- c(NaN, NA)
  expect_identical(is.na(rray_sum(x, 1L)), array(TRUE, 1L))
})
