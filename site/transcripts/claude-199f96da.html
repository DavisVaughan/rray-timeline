r_obj* rray_move_axes(
  r_obj* x,
  r_obj* axes,
  r_obj* to,
  struct rray_arg* arg,
  struct r_lazy error_call
) {
  const int dimensionality = rray_dimensionality(x, arg, error_call);
  check_max_dimensionality(dimensionality);

  axes = KEEP(
    arg_as_axes_unsorted(axes, dimensionality, rray_args.axes, error_call)
  );
  to =
    KEEP(arg_as_axes_unsorted(to, dimensionality, rray_args.to, error_call));

  const r_ssize axes_size = r_length(axes);
  const r_ssize to_size = r_length(to);

  if (axes_size != to_size) {
    r_abort_lazy_call(
      error_call,
      "`axes` and `to` must be the same length, not lengths %" R_PRIdXLEN_T
      " and %" R_PRIdXLEN_T ".",
      axes_size,
      to_size
    );
  }

  const int* v_axes = r_int_cbegin(axes);
  const int* v_to = r_int_cbegin(to);

  r_obj* permutation = KEEP(r_alloc_integer(dimensionality));
  int* v_permutation = r_int_begin(permutation);

  bool v_moved[RRAY_MAX_DIMENSIONALITY] = {false};
  bool v_filled[RRAY_MAX_DIMENSIONALITY] = {false};

  for (r_ssize i = 0; i < axes_size; ++i) {
    const int axis = v_axes[i];
    const int location = v_to[i];

    v_permutation[location - 1] = axis;
    v_moved[axis - 1] = true;
    v_filled[location - 1] = true;
  }

  int axis = 1;

  for (int i = 0; i < dimensionality; ++i) {
    if (v_filled[i]) {
      continue;
    }

    while (v_moved[axis - 1]) {
      ++axis;
    }

    v_permutation[i] = axis;
    ++axis;
  }

  r_obj* out = KEEP(rray_permute_axes(x, permutation, arg, error_call));

  FREE(4);
  return out;
}

#' `rray_move_axes()` moves `axes` to new positions. The axes that don't move

#'

#'
#' @param x An array.
#'

#'   `x` more than once.
#'
#' @param to An integer vector of positions to move `axes` to. `axes[[i]]` ends
#'   up at position `to[[i]]`. It must be the same length as `axes`, and must
#'   not use a position more than once.
#'
#' @returns
#' An array with the `axes` of `x` moved to the positions in `to`.
#'
#' @seealso [rray_permute_axes()]
#'
#' @export
#' @examples
#' x <- array(1:24, c(2, 3, 4))
#'
#' # Move the first axis to the end
#' # (2, 3, 4) -> (3, 4, 2)

#'
#' # Move the last axis to the front
#' # (2, 3, 4) -> (4, 2, 3)

#'
#' # Swap the first two axes
#' # (2, 3, 4) -> (3, 2, 4)
#' rray_dimensions(rray_move_axes(x, c(1, 2), c(2, 1)))

How it works

rray_move_axes() validates, builds a permutation, and calls rray_permute_axes() in C. The permutation is built directly: put axes[[i]] at position to[[i]], then fill the leftover positions with the leftover axes in ascending order. That matches numpy's moveaxis() on every case I checked, including multi-axis moves like rray_move_axes(x, c(1, 2), c(4, 3)) on a 4D array.

Neither. The leftovers don't go as a block to one side of the moved axes: the moved axes claim exact positions, and the leftovers pour into the holes that remain, left to right, keeping their relative order.

Two steps:

  1. For each i, put axes[[i]] at position to[[i]]. These are absolute positions in the output, not offsets from anything.

  2. Walk the output positions from 1 upward. Every position not claimed in step 1 takes the next unmoved axis, in ascending axis order.

Here is x with dims (1, 2, 3, 4), so each dimension value tells you which source axis it came from:

rray_move_axes(x, 2, 4)           # 1 3 4 2
rray_move_axes(x, 2, 1)           # 2 1 3 4
rray_move_axes(x, c(4, 1), c(2, 3))  # 2 4 1 3

Take the first one. Axis 2 is pinned to position 4. Holes at 1, 2, 3, and the unmoved axes 1, 3, 4 drop in that order, giving 1 3 4 2.

The third shows why "left or right" doesn't describe it. Axis 4 goes to position 2 and axis 1 to position 3, leaving holes at 1 and 4. Unmoved axes 2 and 3 fill them in order, so axis 2 lands to the left of both moved axes and axis 3 to the right. They were split up, not pushed to one side.

This is the same rule as numpy's moveaxis().
