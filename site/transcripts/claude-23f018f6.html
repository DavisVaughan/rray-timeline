r_obj* rray_set_titles(
  r_obj* x,
  r_obj* titles,
  struct rray_arg* arg,
  struct r_lazy error_call
) {
  check_unclassed(x, arg, error_call);
  x = KEEP(arg_as_array(x, arg, error_call));

  if (titles == r_null) {
    r_obj* out = rray_clear_titles(x);
    FREE(1);
    return out;
  }

  const int dimensionality = rray_dimensionality(x, arg, error_call);

  if (r_typeof(titles) != R_TYPE_character) {
    r_abort_lazy_call(
      error_call,
      "%s must be a character vector or `NULL`, not %s.",
      rray_arg_format(rray_args.titles),
      r_obj_type_friendly(titles)
    );
  }

  if (r_length(titles) != dimensionality) {
    r_abort_lazy_call(
      error_call,
      "%s must have length %d to match the dimensionality of %s, "
      "not length %" R_PRI_SSIZE ".",
      rray_arg_format(rray_args.titles),
      dimensionality,
      rray_arg_format(arg),
      r_length(titles)
    );
  }

  r_obj* old_names = r_dim_names(x);
  r_obj* new_names = KEEP(r_alloc_list(dimensionality));

  if (old_names != r_null) {
    r_obj* const* v_old_names = r_list_cbegin(old_names);
    for (int i = 0; i < dimensionality; ++i) {
      r_list_poke(new_names, i, v_old_names[i]);
    }
  }

  r_attrib_poke_names(new_names, titles);

  r_obj* out = KEEP(r_wrap(x));
  r_attrib_poke_dim_names(out, new_names);

  FREE(3);
  return out;
}

r_obj* rray_set_axis_title(
  r_obj* x,
  int axis,
  r_obj* title,
  struct rray_arg* arg,
  struct r_lazy error_call
) {
  check_unclassed(x, arg, error_call);
  x = KEEP(arg_as_array(x, arg, error_call));

  const int dimensionality = rray_dimensionality(x, arg, error_call);
  check_axis(axis, dimensionality, rray_args.axis, error_call);

  if (title == r_null) {
    r_obj* out = rray_clear_axis_title(x, axis, arg, error_call);
    FREE(1);
    return out;
  }

  if (r_typeof(title) != R_TYPE_character) {
    r_abort_lazy_call(
      error_call,
      "%s must be a character vector or `NULL`, not %s.",
      rray_arg_format(rray_args.title),
      r_obj_type_friendly(title)
    );
  }

  if (r_length(title) != 1) {
    r_abort_lazy_call(
      error_call,
      "%s must have length 1, not length %" R_PRI_SSIZE ".",
      rray_arg_format(rray_args.title),
      r_length(title)
    );
  }

  r_obj* names = r_dim_names(x);
  r_obj* old_titles = names == r_null ? r_null : r_names(names);

  r_obj* new_titles = KEEP(r_alloc_character(dimensionality));

  if (old_titles != r_null) {
    r_obj* const* v_old_titles = r_chr_cbegin(old_titles);
    for (int i = 0; i < dimensionality; ++i) {
      r_chr_poke(new_titles, i, v_old_titles[i]);
    }
  }

  r_chr_poke(new_titles, axis - 1, r_chr_get(title, 0));

  r_obj* out = KEEP(rray_set_titles(x, new_titles, arg, error_call));

  FREE(3);
  return out;
}

static r_obj* rray_clear_titles(r_obj* x) {

  r_obj* out = KEEP(r_wrap(x));
  r_attrib_poke_dim_names(out, new_names);

  FREE(2);
  return out;
}

You're right. I removed both calls from rray_axis_title() in src/titles.c.

They weren't needed because rray_axis_title() never uses x directly. It only passes x to two helpers, and both already run the same checks:

  • rray_dimensionality() calls rray_dimensions(), which runs check_unclassed() and arg_as_array() first. So a classed input still fails before check_axis() runs, and the error is the same.

  • rray_titles() calls rray_names(), which does the same two steps.

With the converted x gone, FREE(2) is now FREE(1). titles is still protected while r_str_as_character() allocates.

I ran clang-format, and the titles tests pass (51 passing, 0 failing). Nothing is committed.

rray_axis_names() in src/names.c has the same two extra calls for the same reason (it goes through rray_dimensionality() and rray_names()). Want me to remove them there too?
